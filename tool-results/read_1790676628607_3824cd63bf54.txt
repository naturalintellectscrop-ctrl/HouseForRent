    24→export async function freshnessWindowDays(): Promise<number> {
    25→  const row = await db.configParameter.findUnique({ where: { key: 'freshness_window_days' } });
    26→  const parsed = row ? parseInt(row.value, 10) : NaN;
    27→  return Number.isFinite(parsed) && parsed > 0 ? parsed : DEFAULT_FRESHNESS_DAYS;
    28→}

    30→export async function setConfigParameter(params: { key: string; value: string; valueType: string; changedBy: string }) {
    31→  const existing = await db.configParameter.findUnique({ where: { key: params.key } });
    32→  if (existing) {
    33→    const [updated] = await db.$transaction([
    34→      db.configParameter.update({ where: { key: params.key }, data: { value: params.value, valueType: params.valueType } }),
    35→      db.configVersion.create({
    36→        data: { parameterId: existing.id, value: params.value, changedBy: params.changedBy },
    37→      }),
    38→    ]);
    39→    return updated;
    40→  }
    41→  const created = await db.configParameter.create({
    42→    data: { key: params.key, value: params.value, valueType: params.valueType },
    43→  });
    44→  await db.configVersion.create({
    45→    data: { parameterId: created.id, value: params.value, changedBy: params.changedBy },
    46→  });
    47→  return created;
    48→}

    52→export interface WithFreshness<T extends { availabilityConfirmedAt: Date | null }> {
    53→  item: T;
    54→  isStale: boolean;
    55→  daysSinceConfirmed: number | null;
    56→}
    57→
    58→/**
    59→ * Staleness is computed, not stored (FR-2.3). A listing whose availability
    60→ * was never confirmed is treated as stale — absence of evidence is not
    61→ * evidence of availability.
    62→ */
    63→export function withFreshness<T extends { availabilityConfirmedAt: Date | null }>(
    64→  listing: T,
    65→  windowDays: number,
    66→  asOf: Date = new Date(),
    67→): WithFreshness<T> {
    68→  if (!listing.availabilityConfirmedAt) {
    69→    return { item: listing, isStale: true, daysSinceConfirmed: null };
    70→  }
    71→  const elapsedMs = asOf.getTime() - listing.availabilityConfirmedAt.getTime();
    72→  const daysSinceConfirmed = Math.floor(elapsedMs / (24 * 60 * 60 * 1000));
    73→  return { item: listing, isStale: daysSinceConfirmed > windowDays, daysSinceConfirmed };
    74→}

   149→const DEFAULT_LIMIT = 24;
   150→const MAX_LIMIT = 60;
   151→
   152→/**
   153→ * Corridor-scoped tenant search (FR-4.1). The three NON-NEGOTIABLE
   154→ * constraints are applied regardless of filters — they are what the public
   155→ * feed MEANS. `filters` only narrows.
   156→ */
   157→export async function publicSearch(filters: SearchFilters = {}, asOf: Date = new Date()): Promise<SearchResponse> {
   158→  const now = asOf;
   159→  const windowDays = await freshnessWindowDays();
   160→  const staleCutoff = new Date(now.getTime() - windowDays * 24 * 60 * 60 * 1000);
   161→
   162→  const where: Prisma.ListingWhereInput = {
   163→    publicationState: 'live',
   164→    verificationState: 'verified',
   165→    property: { neighbourhood: { inServiceArea: true } },
   166→    // Stale listings leave search rather than quietly becoming a wasted trip.
   167→    availabilityConfirmedAt: filters.includeStale ? undefined : { gte: staleCutoff },
   168→    availabilityStatus: 'available',
   169→    ...(filters.neighbourhoodIds?.length
   170→      ? { property: { neighbourhood: { inServiceArea: true, id: { in: filters.neighbourhoodIds } } } }
   171→      : {}),
   172→    ...(filters.minRent !== undefined || filters.maxRent !== undefined
   173→      ? { monthlyRent: { gte: filters.minRent, lte: filters.maxRent } }
   174→      : {}),
   175→    ...(filters.bedrooms !== undefined ? { property: { bedrooms: filters.bedrooms } } : {}),
   176→    ...(filters.furnished ? { property: { furnished: filters.furnished } } : {}),
   177→    ...(filters.propertyType ? { property: { propertyType: filters.propertyType } } : {}),
   178→  };
   179→
   180→  if (filters.q) {
   181→    const q = filters.q.trim();
   182→    if (q) {
   183→      where.OR = [
   184→        { property: { neighbourhood: { name: { contains: q } } } },
   185→        { property: { neighbourhood: { district: { contains: q } } } },
   186→        { property: { landmarkText: { contains: q } } },
   187→      ];
   188→    }
   189→  }
   190→
   191→  const limit = Math.min(filters.limit ?? DEFAULT_LIMIT, MAX_LIMIT);
   192→  const offset = filters.offset ?? 0;
   193→
   194→  const orderBy: Prisma.ListingOrderByWithRelationInput =
   195→    filters.sort === 'rent_asc'
   196→      ? { monthlyRent: 'asc' }
   197→      : filters.sort === 'rent_desc'
   198→        ? { monthlyRent: 'desc' }
   199→        : filters.sort === 'newest'
   200→          ? { createdAt: 'desc' }
   201→          : { availabilityConfirmedAt: 'desc' }; // fresh first — a wasted trip is THE failure
   202→
   203→  const [rows, totalCount] = await Promise.all([
   204→    db.listing.findMany({
   205→      where,
   206→      orderBy,
   207→      take: limit,
   208→      skip: offset,
   209→      include: {
   210→        property: { include: { neighbourhood: true } },
   211→        photos: { orderBy: { position: 'asc' }, include: { asset: true } },
   212→      },
   213→    }),
   214→    db.listing.count({ where }),
   215→  ]);
   216→
   217→  const results: SearchResult[] = rows.map((l) => {
   218→    const f = withFreshness(l, windowDays, now);
   219→    return {
   220→      listingId: l.id,
   221→      propertyId: l.propertyId,
   222→      monthlyRent: l.monthlyRent.toString(),
   223→      requiredMonthsUpfront: l.requiredMonthsUpfront,
   224→      depositAmount: l.depositAmount.toString(),
   225→      bedrooms: l.property.bedrooms,
   226→      bathrooms: l.property.bathrooms,
   227→      propertyType: l.property.propertyType,
   228→      furnished: l.property.furnished,
   229→      neighbourhoodName: l.property.neighbourhood.name,
   230→      landmarkText: l.property.landmarkText,
   231→      isVerified: l.verificationState === 'verified',
   232→      isStale: f.isStale,
   233→      daysSinceConfirmed: f.daysSinceConfirmed,
   234→      photos: l.photos.map(photoToView),
   235→      freeForTenants: true,
   236→    };
   237→  });
   238→
   239→  const emptyStateMessage =
   240→    totalCount === 0
   241→      ? 'No homes match this search right now. House For Rent publishes only homes our officers have visited and confirmed available — a short list somebody stood inside is the product. New verified homes appear as field visits are completed.'
   242→      : totalCount < 5
   243→        ? 'Only a few homes are live in this corridor at the moment. Every one of them has been visited and confirmed by a field officer — that verification is what takes time.'
   244→        : null;
   245→
   246→  return { results, totalCount, limit, offset, emptyStateMessage };
   247→}

   249→/** Public detail. Reuses the feed's visibility rules — no separate opinion. */
   250→export async function publicDetail(listingId: string, asOf: Date = new Date()) {
   251→  const windowDays = await freshnessWindowDays();
   252→  const l = await db.listing.findUnique({
   253→    where: { id: listingId },
   254→    include: {
   255→      property: { include: { neighbourhood: true, owner: { select: { displayName: true } } } },
   256→      photos: { orderBy: { position: 'asc' }, include: { asset: true } },
   257→      listingAmenities: { include: { amenity: true } },
   258→    },
   259→  });
   260→  if (!l) return null;
   261→  if (l.publicationState !== 'live' || l.verificationState !== 'verified') return null;
   262→  if (!l.property.neighbourhood.inServiceArea) return null;
   263→
   264→  const f = withFreshness(l, windowDays, asOf);
   265→  return {
   266→    listingId: l.id,
   267→    propertyId: l.propertyId,
   268→    monthlyRent: l.monthlyRent.toString(),
   269→    requiredMonthsUpfront: l.requiredMonthsUpfront,
   270→    depositAmount: l.depositAmount.toString(),
   271→    /** The server-stated total a tenant will be asked to fund — derived from
   272→     * the same terms fund-escrow derives its authoritative amount from (F-012). */
   273→    expectedUpfront: (l.monthlyRent * BigInt(l.requiredMonthsUpfront) + l.depositAmount).toString(),
   274→    bedrooms: l.property.bedrooms,
   275→    bathrooms: l.property.bathrooms,
   276→    propertyType: l.property.propertyType,
   277→    furnished: l.property.furnished,
   278→    neighbourhoodName: l.property.neighbourhood.name,
   279→    neighbourhoodDistrict: l.property.neighbourhood.district,
   280→    landmarkText: l.property.landmarkText,
   281→    streetAddress: l.property.streetAddress,
   282→    descriptionText: l.descriptionText,
   283→    isVerified: l.verificationState === 'verified',
   284→    isStale: f.isStale,
   285→    daysSinceConfirmed: f.daysSinceConfirmed,
   286→    availabilityConfirmedAt: l.availabilityConfirmedAt,
   287→    amenities: l.listingAmenities.map((a) => a.amenity.name),
   288→    photos: l.photos.map(photoToView),
   289→    listerDisplayName: l.property.owner.displayName,
   290→    freeForTenants: true,
   291→  };
   292→}

   331→        streetAddress: params.streetAddress,

   367→export async function effectiveCommissionRate(asOf: Date = new Date()) {
   368→  const versions = await db.commissionRateVersion.findMany({
   369→    orderBy: { effectiveFrom: 'desc' },
   370→  });
   371→  const current =
   372→    versions.find((v) => v.effectiveFrom.getTime() <= asOf.getTime() && (!v.effectiveTo || v.effectiveTo.getTime() > asOf.getTime())) ??
   373→    versions[0];
   374→  if (!current) {
   375→    return db.commissionRateVersion.create({
   376→      data: { rateBp: 10000, effectiveFrom: new Date('2026-01-01') },
   377→    });
   378→  }
   379→  return current;
   380→}

   437→/**
   438→ * The ONE implementation of "what stands between this listing and the live
   439→ * feed" (the publish gate, expressed as blocker codes). evaluatePublish,
   440→ * findForLister and the admin verification queue all call this — three
   441→ * surfaces, one opinion. `mandateVerified` is resolved by the caller (it
   442→ * needs a (lister, property) query only for non-owner tiers).
   443→ */
   444→function publishBlockers(input: {
   445→  verificationState: string;
   446→  inServiceArea: boolean;
   447→  hasAcceptedAgreement: boolean;
   448→  listerTier: ListerTier | undefined;
   449→  mandateVerified: boolean;
   450→}): string[] {
   451→  const blockedBy: string[] = [];
   452→  if (input.verificationState !== 'verified') blockedBy.push('field_verification');
   453→  if (!input.inServiceArea) blockedBy.push('outside_service_area');
   454→  if (input.listerTier && input.listerTier !== 'property_owner' && !input.mandateVerified) {
   455→    blockedBy.push('mandate');
   456→  }
   457→  if (!input.hasAcceptedAgreement) blockedBy.push('listing_agreement');
   458→  return blockedBy;
   459→}
   460→
   461→/**
   462→ * THE publish gate — all four preconditions enforced server-side (FR-2.5,
   463→ * FR-3.1, FR-3.2, FR-9.1). `dryRun` produces the same evaluation without
   464→ * flipping state, so the landlord UI can show what is still missing.
   465→ */
   466→export async function evaluatePublish(listingId: string, dryRun = true) {
   467→  const listing = await db.listing.findUnique({
   468→    where: { id: listingId },
   469→    include: {
   470→      property: { include: { neighbourhood: true, owner: { include: { listerProfile: true } } } },
   471→      listingAgreements: { where: { accepted: true }, take: 1 },
   472→    },
   473→  });
   474→  if (!listing) throw new Error('listing not found');
   475→
   476→  const tier = listing.property.owner.listerProfile?.tier as ListerTier | undefined;
   477→  const mandateVerified = tier
   478→    ? await canPublish({ listerTier: tier, listerPartyId: listing.property.ownerPartyId, propertyId: listing.propertyId })
   479→    : true;
   480→  const blockedBy = publishBlockers({
   481→    verificationState: listing.verificationState,
   482→    inServiceArea: listing.property.neighbourhood.inServiceArea,
   483→    hasAcceptedAgreement: listing.listingAgreements.length > 0,
   484→    listerTier: tier,
   485→    mandateVerified,
   486→  });
   487→
   488→  if (!dryRun && blockedBy.length === 0) {
   489→    await db.$transaction(async (tx) => {
   490→      await tx.listing.update({ where: { id: listingId }, data: { publicationState: 'live' } });
   491→      await tx.auditEvent.create({
   492→        data: {
   493→          actorPartyId: listing.property.ownerPartyId,
   494→          actorRole: 'lister',
   495→          action: 'listing_published',
   496→          entityType: 'listing',
   497→          entityId: listing.id,
   498→          detail: JSON.stringify({ publicationState: 'live', agreementId: listing.listingAgreements[0].id }),
   499→        },
   500→      });
   501→    });
   502→  }
   503→
   504→  return { blockedBy, canPublish: blockedBy.length === 0 };
   505→}

   507→/** The landlord publishes (only after the gate passes). */
   508→export async function publishListing(params: { listingId: string; listerPartyId: string }) {
   509→  const listing = await db.listing.findUnique({ where: { id: params.listingId }, include: { property: true } });
   510→  if (!listing) throw new Error('listing not found');
   511→  if (listing.property.ownerPartyId !== params.listerPartyId) throw new NotYourListingError();
   512→
   513→  const result = await evaluatePublish(params.listingId, false);
   514→  if (!result.canPublish) {
   515→    throw new Error(`listing cannot be published: blocked by ${result.blockedBy.join(', ')}`);
   516→  }
   517→  return db.listing.findUnique({ where: { id: params.listingId } });
   518→}

   531→export async function findForLister(listerPartyId: string) {
   532→  const listings = await db.listing.findMany({
   533→    where: { property: { ownerPartyId: listerPartyId } },
   534→    include: {
   535→      property: { include: { neighbourhood: true } },
   536→      listingAgreements: { orderBy: { createdAt: 'desc' } },
   537→      photos: { orderBy: { position: 'asc' }, include: { asset: true } },
   538→    },
   539→    orderBy: { createdAt: 'desc' },
   540→  });
   541→
   542→  const tier = (await db.listerProfile.findUnique({ where: { partyId: listerPartyId } }))?.tier;
   543→
   544→  return Promise.all(
   545→    listings.map(async (listing) => {
   546→      const tierCast = tier as ListerTier | undefined;
   547→      const mandateVerified = tierCast
   548→        ? await canPublish({ listerTier: tierCast, listerPartyId, propertyId: listing.propertyId })
   549→        : true;
   550→      const blockedBy = publishBlockers({
   551→        verificationState: listing.verificationState,
   552→        inServiceArea: listing.property.neighbourhood.inServiceArea,
   553→        hasAcceptedAgreement: listing.listingAgreements.some((a) => a.accepted),
   554→        listerTier: tierCast,
   555→        mandateVerified,
   556→      });
   557→      const accepted = listing.listingAgreements.find((a) => a.accepted);
   558→
   559→      return {
   560→        id: listing.id,
   561→        propertyId: listing.propertyId,
   562→        monthlyRent: listing.monthlyRent.toString(),
   563→        depositAmount: listing.depositAmount.toString(),
   564→        requiredMonthsUpfront: listing.requiredMonthsUpfront,
   565→        bedrooms: listing.property.bedrooms,
   566→        bathrooms: listing.property.bathrooms,
   567→        neighbourhoodName: listing.property.neighbourhood.name,
   568→        landmarkText: listing.property.landmarkText,
   569→        verificationState: listing.verificationState,
   570→        publicationState: listing.publicationState,
   571→        availabilityStatus: listing.availabilityStatus,
   572→        availabilityConfirmedAt: listing.availabilityConfirmedAt,
   573→        hasAcceptedAgreement: Boolean(accepted),
   574→        agreementId: accepted?.id ?? null,
   575→        agreementAcceptedAt: accepted?.acceptedAt ?? null,
   576→        commissionRateBp: accepted?.commissionRateVersionId
   577→          ? (await db.commissionRateVersion.findUnique({ where: { id: accepted.commissionRateVersionId } }))?.rateBp ?? null
   578→          : null,
   579→        photos: listing.photos.map(photoToView),
   580→        blockedBy,
   581→        canPublish: blockedBy.length === 0,
   582→      };
   583→    }),
   584→  );
   585→}

   593→/**
   594→ * FR-10.2 — the admin verification queue: every listing not yet live (and
   595→ * not withdrawn), with what each is actually waiting on.
   596→ *
   597→ * The previous implementation hard-coded `verificationState: 'unverified'`,
   598→ * `mandateState: null` and `hasAcceptedAgreement: false` — so the console
   599→ * showed a mandate column that could never hold a value and a queue that
   600→ * could never name its real blockers. This version reads the states and
   601→ * computes `blockedBy` through the same `publishBlockers` gate the publish
   602→ * endpoint enforces, so the console and the gate cannot disagree.
   603→ */
   604→export async function adminVerificationQueue() {
   605→  const listings = await db.listing.findMany({
   606→    where: { publicationState: { in: ['draft', 'awaiting_verification'] } },
   607→    include: {
   608→      property: { include: { neighbourhood: true, owner: { include: { listerProfile: true } } } },
   609→      listingAgreements: { where: { accepted: true }, take: 1 },
   610→    },
   611→    orderBy: { createdAt: 'asc' },
   612→  });
   613→
   614→  return Promise.all(
   615→    listings.map(async (listing) => {
   616→      const tier = listing.property.owner.listerProfile?.tier as ListerTier | undefined;
   617→      const mandateVerified = tier
   618→        ? await canPublish({ listerTier: tier, listerPartyId: listing.property.ownerPartyId, propertyId: listing.propertyId })
   619→        : true;
   620→      const blockedBy = publishBlockers({
   621→        verificationState: listing.verificationState,
   622→        inServiceArea: listing.property.neighbourhood.inServiceArea,
   623→        hasAcceptedAgreement: listing.listingAgreements.length > 0,
   624→        listerTier: tier,
   625→        mandateVerified,
   626→      });
   627→      const mandate = tier && tier !== 'property_owner'
   628→        ? await db.propertyMandate.findFirst({
   629→            where: { propertyId: listing.propertyId, listerPartyId: listing.property.ownerPartyId },
   630→            orderBy: { createdAt: 'desc' },
   631→          })
   632→        : null;
   633→
   634→      return {
   635→        listingId: listing.id,
   636→        propertyId: listing.propertyId,
   637→        listerPartyId: listing.property.ownerPartyId,
   638→        listerName: listing.property.owner.displayName,
   639→        listerTier: tier ?? null,
   640→        neighbourhood: listing.property.neighbourhood.name,
   641→        inServiceArea: listing.property.neighbourhood.inServiceArea,
   642→        verificationState: listing.verificationState,
   643→        publicationState: listing.publicationState,
   644→        mandateState: mandate?.state ?? null,
   645→        hasAcceptedAgreement: listing.listingAgreements.length > 0,
   646→        blockedBy,
   647→        createdAt: listing.createdAt.toISOString(),
   648→      };
   649→    }),
   650→  );
   651→}
