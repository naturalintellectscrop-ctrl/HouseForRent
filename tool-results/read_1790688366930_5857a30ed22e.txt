   379→async function resolveListingsSearch(sp: URLSearchParams): Promise<SearchResponse> {
   380→  const filters: Parameters<typeof publicSearch>[0] = {};
   381→  const nid = sp.get('neighbourhoodId') ?? sp.get('neighbourhoodIds');
   382→  if (nid) filters.neighbourhoodIds = nid.split(',').filter(Boolean);
   383→  const minRent = sp.get('minRent');
   384→  if (minRent && /^[0-9]+$/.test(minRent)) filters.minRent = BigInt(minRent);
   385→  const maxRent = sp.get('maxRent');
   386→  if (maxRent && /^[0-9]+$/.test(maxRent)) filters.maxRent = BigInt(maxRent);
   387→  const bedrooms = sp.get('bedrooms');
   388→  if (bedrooms && /^[0-9]+$/.test(bedrooms)) filters.bedrooms = parseInt(bedrooms, 10);
   389→  const furnished = sp.get('furnished');
   390→  if (furnished) filters.furnished = furnished;
   391→  const propertyType = sp.get('propertyType');
   392→  if (propertyType) filters.propertyType = propertyType as never;
   393→  const q = sp.get('q');
   394→  if (q) filters.q = q;
   395→  const sort = sp.get('sort');
   396→  if (sort && ['fresh', 'rent_asc', 'rent_desc', 'newest'].includes(sort)) {
   397→    filters.sort = sort as never;
   398→  }
   399→  const limit = sp.get('limit');
   400→  if (limit && /^[0-9]+$/.test(limit)) filters.limit = parseInt(limit, 10);
   401→  const offset = sp.get('offset');
   402→  if (offset && /^[0-9]+$/.test(offset)) filters.offset = parseInt(offset, 10);
   403→  return publicSearch(filters);
   404→}

   406→async function resolveListingDetail(id: string) {
   407→  const detail = await publicDetail(id);
   408→  if (!detail) throw new ApiError(404, 'LISTING_NOT_FOUND', 'that listing does not exist or is not public');
   409→  // The officer's structured verdict on this home, when one exists — the
   410→  // same record the verification queue writes.
   411→  const lastConducted = await db.viewing.findFirst({
   412→    where: { listingId: id, status: 'conducted', fieldReport: { isNot: null } },
   413→    orderBy: { updatedAt: 'desc' },
   414→    include: { fieldReport: true },
   415→  });
   416→  const fieldConfirmed = lastConducted?.fieldReport
   417→    ? {
   418→        conditionRating: lastConducted.fieldReport.conditionRating as 'excellent' | 'good' | 'fair' | 'poor',
   419→        matchesListing: lastConducted.fieldReport.matchesListing,
   420→        isAvailable: lastConducted.fieldReport.isAvailable,
   421→        reportedAt: lastConducted.fieldReport.reportedAt.toISOString(),
   422→      }
   423→    : null;
   424→  // The caller's bookmark state, only when the caller is a signed-in tenant
   425→  // (saving is a tenant surface — anonymous visitors get no toggle).
   426→  const caller = await resolveSession();
   427→  const savedByCaller = caller?.role === 'tenant' ? await isSaved(caller.partyId, id) : null;
   428→  return {
   429→    ...detail,
   430→    neighbourhoodId: '',
   431→    geoLat: null,
   432→    geoLng: null,
   433→    amenities: detail.amenities.map((name, i) => ({ id: String(i), name })),
   434→    fieldConfirmed,
   435→    savedByCaller,
   436→  };
   437→}

  1075→type Resolver = () => Promise<unknown>;
  1076→
  1077→/**
  1078→ * Resolves an API path in-process. Route ORDER MATTERS: `/v1/listings/mine`
  1079→ * must be tried before `/v1/listings/:id`, or "mine" would be read as an id.
  1080→ */
  1081→function resolverFor(path: string, authenticated: boolean): Resolver {
  1082→  const u = new URL(path, 'http://in-process');
  1083→  const parts = u.pathname.split('/').filter(Boolean); // ['v1', ...]
  1084→
  1085→  // public reads
  1086→  if (parts[1] === 'listings' && parts.length === 2) return () => resolveListingsSearch(u.searchParams);
  1087→  if (parts[1] === 'listings' && parts[2] === 'mine') return () => resolveMyListings();
  1088→  // saved must precede /:id — same ordering rule as /mine above.
  1089→  if (parts[1] === 'listings' && parts[2] === 'saved') return () => resolveSavedListings();
  1090→  if (parts[1] === 'listings' && parts.length === 3) return () => resolveListingDetail(parts[2]);
  1091→  if (parts[1] === 'listings' && parts[3] === 'agreement') return () => resolvePresentedTerms(parts[2]);
  1092→  if (parts[1] === 'listings' && parts[3] === 'photos' && parts.length === 4) return () => resolveListingPhotos(parts[2]);
  1093→  if (parts[1] === 'neighbourhoods') return () => resolveNeighbourhoods();
  1094→  if (parts[1] === 'commission-rate') return () => resolveCommissionRate();
  1095→
  1096→  // authenticated reads
  1097→  if (!authenticated) return () => { throw unauthenticated(); };
  1098→  if (parts[1] === 'viewings' && parts[2] === 'mine') return () => resolveMyViewings();
  1099→  if (parts[1] === 'viewings' && parts[2] === 'assigned' && parts[3] === 'me') return () => resolveAssignedViewings();
  1100→  if (parts[1] === 'viewings' && parts[2] === 'dispatch-queue') return () => resolveDispatchQueue();
  1101→  if (parts[1] === 'viewings' && parts[2] === 'for-lister') return () => resolveListerViewings();
  1102→  // TASK 7-b (additive): filtered evidence lookup — must precede the
  1103→  // unfiltered introductions route below.
  1104→  if (parts[1] === 'viewings' && parts[2] === 'introductions' && hasIntroductionFilters(u.searchParams)) {
  1105→    return () => resolveIntroductionsFiltered(u.searchParams);
  1106→  }
  1107→  if (parts[1] === 'viewings' && parts[2] === 'introductions') return () => resolveIntroductions();
  1108→  if (parts[1] === 'viewings' && parts.length === 3) return () => resolveViewingDetail(parts[2]);
  1109→  if (parts[1] === 'identity' && parts[2] === 'me') return () => resolveIdentityMe();
  1110→  if (parts[1] === 'activity' && parts[2] === 'mine') return () => resolveActivityMine();
  1111→  if (parts[1] === 'deals' && parts.length === 2) return () => resolvePartyDeals();
  1112→  if (parts[1] === 'deals' && parts.length === 3) return () => resolveDealDetail(parts[2]);
  1113→  if (parts[1] === 'landlord' && parts[2] === 'financials') return () => resolveLandlordFinancials();
  1114→  // Task 10-b (additive): the mandate flow's reads. The decision POST is a
  1115→  // mutation and belongs to its route handler (which calls the service
  1116→  // directly), never to this GET-only dispatcher.
  1117→  if (parts[1] === 'landlord' && parts[2] === 'mandates') return () => resolveLandlordMandates();
  1118→  if (parts[1] === 'admin' && parts[2] === 'mandates') return () => resolveAdminMandates(u.searchParams);
  1119→  if (parts[1] === 'admin' && parts[2] === 'launch-gate') return () => resolveLaunchGate();
  1120→  if (parts[1] === 'admin' && parts[2] === 'verification-queue') return () => resolveVerificationQueue();
  1121→  if (parts[1] === 'admin' && parts[2] === 'users' && parts.length === 4) {
  1122→    return () => resolveAdminPartyDetail(parts[3]);
  1123→  }
  1124→  if (parts[1] === 'admin' && parts[2] === 'users') return () => resolveAdminUsers(u.searchParams);
  1125→  if (parts[1] === 'ops' && parts[2] === 'neighbourhoods') return () => resolveOpsNeighbourhoods();
  1126→  // Task 16: who dispatch can send — the reassign control's select data.
  1127→  if (parts[1] === 'ops' && parts[2] === 'officers') return () => resolveOpsOfficers();
  1128→  // TASK 7-b (additive): the deal queue's "Show only" filter passes ?status=
  1129→  // — must precede the unfiltered admin deals route below. The distribution
  1130→  // stays whole (it is the shape of the whole book); only the rows narrow.
  1131→  if (parts[1] === 'admin' && parts[2] === 'deals' && u.searchParams.has('status')) {
  1132→    return async () => {
  1133→      const status = u.searchParams.get('status') ?? '';
  1134→      const states = await resolveDealStates();
  1135→      const rows = states.rows.filter((r) => r.status === status);
  1136→      return { ...states, total: rows.length, rows };
  1137→    };
  1138→  }
  1139→  if (parts[1] === 'admin' && parts[2] === 'deals') return () => resolveDealStates();
  1140→  if (parts[1] === 'admin' && parts[2] === 'reconciliation') return () => resolveReconciliation();
  1141→  if (parts[1] === 'admin' && parts[2] === 'audit' && parts.length === 4) {
  1142→    return async () => toAuditEvents(await recentAuditEvents(50)).filter((e) => e.subjectRef === parts[3]);
  1143→  }
  1144→  if (parts[1] === 'admin' && parts[2] === 'audit') return async () => toAuditEvents(await recentAuditEvents(100));
  1145→  if (parts[1] === 'admin' && parts[2] === 'commission-rates') return () => resolveCommissionRates();
  1146→  if (parts[1] === 'admin' && parts[2] === 'config' && parts[4] === 'versions') return () => resolveConfigVersions(parts[3]);
  1147→
  1148→  throw new ApiError(404, 'UNKNOWN_PATH', `no in-process resolver for GET ${path}`);
  1149→}

  1151→/**
  1152→ * An unauthenticated GET against the public API. The marketplace is
  1153→ * browsable without an account (Decision 3).
  1154→ */
  1155→export async function apiGet<T>(path: string, _opts: { revalidate?: number } = {}): Promise<T> {
  1156→  return resolverFor(path, false)() as Promise<T>;
  1157→}

  1159→/**
  1160→ * Authenticated GET, in-process. Runs the role check here that the API's
  1161→ * guards would run — same rules, one implementation.
  1162→ *
  1163→ * MUTATIONS DO NOT BELONG HERE: any POST/PUT goes through `postJson()`
  1164→ * from '@/lib/client' in a CLIENT component, over real HTTP to /api/v1 —
  1165→ * this module is server-only (it imports the database), and every state
  1166→ * change must cross an API boundary.
  1167→ */
  1168→export async function api<T>(path: string, init?: { method?: string; body?: Record<string, unknown> }): Promise<T> {
  1169→  const method = init?.method ?? 'GET';
  1170→  if (method !== 'GET') {
  1171→    throw new ApiError(
  1172→      500,
  1173→      'MUTATION_NOT_ALLOWED',
  1174→      'server-side mutations must go through /api/v1 route handlers via postJson()',
  1175→    );
  1176→  }
  1177→  return resolverFor(path, true)() as Promise<T>;
  1178→}
