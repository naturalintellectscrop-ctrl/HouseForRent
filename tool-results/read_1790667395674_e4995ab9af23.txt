   322→   322→async function resolveListingsSearch(sp: URLSearchParams): Promise<SearchResponse> {
   323→   323→  const filters: Parameters<typeof publicSearch>[0] = {};
   324→   324→  const nid = sp.get('neighbourhoodId') ?? sp.get('neighbourhoodIds');
   325→   325→  if (nid) filters.neighbourhoodIds = nid.split(',').filter(Boolean);
   326→   326→  const minRent = sp.get('minRent');
   327→   327→  if (minRent && /^[0-9]+$/.test(minRent)) filters.minRent = BigInt(minRent);
   328→   328→  const maxRent = sp.get('maxRent');
   329→   329→  if (maxRent && /^[0-9]+$/.test(maxRent)) filters.maxRent = BigInt(maxRent);
   330→   330→  const bedrooms = sp.get('bedrooms');
   331→   331→  if (bedrooms && /^[0-9]+$/.test(bedrooms)) filters.bedrooms = parseInt(bedrooms, 10);
   332→   332→  const furnished = sp.get('furnished');
   333→   333→  if (furnished) filters.furnished = furnished;
   334→   334→  const propertyType = sp.get('propertyType');
   335→   335→  if (propertyType) filters.propertyType = propertyType as never;
   336→   336→  const q = sp.get('q');
   337→   337→  if (q) filters.q = q;
   338→   338→  const sort = sp.get('sort');
   339→   339→  if (sort && ['fresh', 'rent_asc', 'rent_desc', 'newest'].includes(sort)) {
   340→   340→    filters.sort = sort as never;
   341→   341→  }
   342→   342→  const limit = sp.get('limit');
   343→   343→  if (limit && /^[0-9]+$/.test(limit)) filters.limit = parseInt(limit, 10);
   344→   344→  const offset = sp.get('offset');
   345→   345→  if (offset && /^[0-9]+$/.test(offset)) filters.offset = parseInt(offset, 10);
   346→   346→  return publicSearch(filters);
   347→   347→}

   349→   349→async function resolveListingDetail(id: string) {
   350→   350→  const detail = await publicDetail(id);
   351→   351→  if (!detail) throw new ApiError(404, 'LISTING_NOT_FOUND', 'that listing does not exist or is not public');
   352→   352→  // The officer's structured verdict on this home, when one exists — the
   353→   353→  // same record the verification queue writes.
   354→   354→  const lastConducted = await db.viewing.findFirst({
   355→   355→    where: { listingId: id, status: 'conducted', fieldReport: { isNot: null } },
   356→   356→    orderBy: { updatedAt: 'desc' },
   357→   357→    include: { fieldReport: true },
   358→   358→  });
   359→   359→  const fieldConfirmed = lastConducted?.fieldReport
   360→   360→    ? {
   361→   361→        conditionRating: lastConducted.fieldReport.conditionRating as 'excellent' | 'good' | 'fair' | 'poor',
   362→   362→        matchesListing: lastConducted.fieldReport.matchesListing,
   363→   363→        isAvailable: lastConducted.fieldReport.isAvailable,
   364→   364→        reportedAt: lastConducted.fieldReport.reportedAt.toISOString(),
   365→   365→      }
   366→   366→    : null;
   367→   367→  return {
   368→   368→    ...detail,
   369→   369→    neighbourhoodId: '',
   370→   370→    geoLat: null,
   371→   371→    geoLng: null,
   372→   372→    amenities: detail.amenities.map((name, i) => ({ id: String(i), name })),

   844→   844→type Resolver = () => Promise<unknown>;
   845→   845→
   846→   846→/**
   847→   847→ * Resolves an API path in-process. Route ORDER MATTERS: `/v1/listings/mine`
   848→   848→ * must be tried before `/v1/listings/:id`, or "mine" would be read as an id.
   849→   849→ */
   850→   850→function resolverFor(path: string, authenticated: boolean): Resolver {
   851→   851→  const u = new URL(path, 'http://in-process');
   852→   852→  const parts = u.pathname.split('/').filter(Boolean); // ['v1', ...]
   853→   853→
   854→   854→  // public reads
   855→   855→  if (parts[1] === 'listings' && parts.length === 2) return () => resolveListingsSearch(u.searchParams);
   856→   856→  if (parts[1] === 'listings' && parts[2] === 'mine') return () => resolveMyListings();
   857→   857→  if (parts[1] === 'listings' && parts.length === 3) return () => resolveListingDetail(parts[2]);
   858→   858→  if (parts[1] === 'listings' && parts[3] === 'agreement') return () => resolvePresentedTerms(parts[2]);
   859→   859→  if (parts[1] === 'listings' && parts[3] === 'photos' && parts.length === 4) return () => resolveListingPhotos(parts[2]);
   860→   860→  if (parts[1] === 'neighbourhoods') return () => resolveNeighbourhoods();
   861→   861→  if (parts[1] === 'commission-rate') return () => resolveCommissionRate();
   862→   862→
   863→   863→  // authenticated reads
   864→   864→  if (!authenticated) return () => { throw unauthenticated(); };
   865→   865→  if (parts[1] === 'viewings' && parts[2] === 'mine') return () => resolveMyViewings();
   866→   866→  if (parts[1] === 'viewings' && parts[2] === 'assigned' && parts[3] === 'me') return () => resolveAssignedViewings();
   867→   867→  if (parts[1] === 'viewings' && parts[2] === 'dispatch-queue') return () => resolveDispatchQueue();
   868→   868→  if (parts[1] === 'viewings' && parts[2] === 'introductions') return () => resolveIntroductions();
   869→   869→  if (parts[1] === 'viewings' && parts.length === 3) return () => resolveViewingDetail(parts[2]);
   870→   870→  if (parts[1] === 'identity' && parts[2] === 'me') return () => resolveIdentityMe();
   871→   871→  if (parts[1] === 'deals' && parts.length === 2) return () => resolvePartyDeals();
   872→   872→  if (parts[1] === 'deals' && parts.length === 3) return () => resolveDealDetail(parts[2]);
   873→   873→  if (parts[1] === 'admin' && parts[2] === 'launch-gate') return () => resolveLaunchGate();
   874→   874→  if (parts[1] === 'admin' && parts[2] === 'verification-queue') return () => resolveVerificationQueue();
   875→   875→  if (parts[1] === 'admin' && parts[2] === 'deals') return () => resolveDealStates();
   876→   876→  if (parts[1] === 'admin' && parts[2] === 'reconciliation') return () => resolveReconciliation();
   877→   877→  if (parts[1] === 'admin' && parts[2] === 'audit' && parts.length === 4) {
   878→   878→    return async () => toAuditEvents(await recentAuditEvents(50)).filter((e) => e.subjectRef === parts[3]);
   879→   879→  }
   880→   880→  if (parts[1] === 'admin' && parts[2] === 'audit') return async () => toAuditEvents(await recentAuditEvents(100));
   881→   881→  if (parts[1] === 'admin' && parts[2] === 'commission-rates') return () => resolveCommissionRates();
   882→   882→  if (parts[1] === 'admin' && parts[2] === 'config' && parts[4] === 'versions') return () => resolveConfigVersions(parts[3]);
   883→   883→
   884→   884→  throw new ApiError(404, 'UNKNOWN_PATH', `no in-process resolver for GET ${path}`);
   885→   885→}

   888→   888→ * An unauthenticated GET against the public API. The marketplace is

   891→   891→export async function apiGet<T>(path: string, _opts: { revalidate?: number } = {}): Promise<T> {

   895→   895→/**
   896→   896→ * Authenticated GET, in-process. Runs the role check here that the API's
   897→   897→ * guards would run — same rules, one implementation.
   898→   898→ *
   899→   899→ * MUTATIONS DO NOT BELONG HERE: any POST/PUT goes through `postJson()`
   900→   900→ * from '@/lib/client' in a CLIENT component, over real HTTP to /api/v1 —
   901→   901→ * this module is server-only (it imports the database), and every state
   902→   902→ * change must cross an API boundary.
   903→   903→ */
   904→   904→export async function api<T>(path: string, init?: { method?: string; body?: Record<string, unknown> }): Promise<T> {

   907→   907→    throw new ApiError(
   908→   908→      500,
   909→   909→      'MUTATION_NOT_ALLOWED',
   910→   910→      'server-side mutations must go through /api/v1 route handlers via postJson()',
   911→   911→    );
   912→   912→  }
   913→   913→  return resolverFor(path, true)() as Promise<T>;
   914→   914→}
