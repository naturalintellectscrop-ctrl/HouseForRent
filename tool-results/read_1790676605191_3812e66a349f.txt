   340→   340→async function resolveListingsSearch(sp: URLSearchParams): Promise<SearchResponse> {
   341→   341→  const filters: Parameters<typeof publicSearch>[0] = {};
   342→   342→  const nid = sp.get('neighbourhoodId') ?? sp.get('neighbourhoodIds');
   343→   343→  if (nid) filters.neighbourhoodIds = nid.split(',').filter(Boolean);
   344→   344→  const minRent = sp.get('minRent');
   345→   345→  if (minRent && /^[0-9]+$/.test(minRent)) filters.minRent = BigInt(minRent);
   346→   346→  const maxRent = sp.get('maxRent');
   347→   347→  if (maxRent && /^[0-9]+$/.test(maxRent)) filters.maxRent = BigInt(maxRent);
   348→   348→  const bedrooms = sp.get('bedrooms');
   349→   349→  if (bedrooms && /^[0-9]+$/.test(bedrooms)) filters.bedrooms = parseInt(bedrooms, 10);
   350→   350→  const furnished = sp.get('furnished');
   351→   351→  if (furnished) filters.furnished = furnished;
   352→   352→  const propertyType = sp.get('propertyType');
   353→   353→  if (propertyType) filters.propertyType = propertyType as never;
   354→   354→  const q = sp.get('q');
   355→   355→  if (q) filters.q = q;
   356→   356→  const sort = sp.get('sort');
   357→   357→  if (sort && ['fresh', 'rent_asc', 'rent_desc', 'newest'].includes(sort)) {
   358→   358→    filters.sort = sort as never;
   359→   359→  }
   360→   360→  const limit = sp.get('limit');
   361→   361→  if (limit && /^[0-9]+$/.test(limit)) filters.limit = parseInt(limit, 10);
   362→   362→  const offset = sp.get('offset');
   363→   363→  if (offset && /^[0-9]+$/.test(offset)) filters.offset = parseInt(offset, 10);
   364→   364→  return publicSearch(filters);
   365→   365→}

   367→   367→async function resolveListingDetail(id: string) {
   368→   368→  const detail = await publicDetail(id);
   369→   369→  if (!detail) throw new ApiError(404, 'LISTING_NOT_FOUND', 'that listing does not exist or is not public');
   370→   370→  // The officer's structured verdict on this home, when one exists — the
   371→   371→  // same record the verification queue writes.
   372→   372→  const lastConducted = await db.viewing.findFirst({
   373→   373→    where: { listingId: id, status: 'conducted', fieldReport: { isNot: null } },
   374→   374→    orderBy: { updatedAt: 'desc' },
   375→   375→    include: { fieldReport: true },
   376→   376→  });
   377→   377→  const fieldConfirmed = lastConducted?.fieldReport
   378→   378→    ? {
   379→   379→        conditionRating: lastConducted.fieldReport.conditionRating as 'excellent' | 'good' | 'fair' | 'poor',
   380→   380→        matchesListing: lastConducted.fieldReport.matchesListing,
   381→   381→        isAvailable: lastConducted.fieldReport.isAvailable,
   382→   382→        reportedAt: lastConducted.fieldReport.reportedAt.toISOString(),
   383→   383→      }
   384→   384→    : null;
   385→   385→  // The caller's bookmark state, only when the caller is a signed-in tenant
   386→   386→  // (saving is a tenant surface — anonymous visitors get no toggle).
   387→   387→  const caller = await resolveSession();
   388→   388→  const savedByCaller = caller?.role === 'tenant' ? await isSaved(caller.partyId, id) : null;
   389→   389→  return {
   390→   390→    ...detail,
   391→   391→    neighbourhoodId: '',
   392→   392→    geoLat: null,
   393→   393→    geoLng: null,
   394→   394→    amenities: detail.amenities.map((name, i) => ({ id: String(i), name })),
   395→   395→    fieldConfirmed,
   396→   396→    savedByCaller,
   397→   397→  };
   398→   398→}

   938→   938→ * must be tried before `/v1/listings/:id`, or "mine" would be read as an id.
   939→   939→ */
   940→   940→function resolverFor(path: string, authenticated: boolean): Resolver {
   941→   941→  const u = new URL(path, 'http://in-process');
   942→   942→  const parts = u.pathname.split('/').filter(Boolean); // ['v1', ...]
   943→   943→
   944→   944→  // public reads
   945→   945→  if (parts[1] === 'listings' && parts.length === 2) return () => resolveListingsSearch(u.searchParams);
   946→   946→  if (parts[1] === 'listings' && parts[2] === 'mine') return () => resolveMyListings();
   947→   947→  // saved must precede /:id — same ordering rule as /mine above.
   948→   948→  if (parts[1] === 'listings' && parts[2] === 'saved') return () => resolveSavedListings();
   949→   949→  if (parts[1] === 'listings' && parts.length === 3) return () => resolveListingDetail(parts[2]);
   950→   950→  if (parts[1] === 'listings' && parts[3] === 'agreement') return () => resolvePresentedTerms(parts[2]);
   951→   951→  if (parts[1] === 'listings' && parts[3] === 'photos' && parts.length === 4) return () => resolveListingPhotos(parts[2]);
   952→   952→  if (parts[1] === 'neighbourhoods') return () => resolveNeighbourhoods();
   953→   953→  if (parts[1] === 'commission-rate') return () => resolveCommissionRate();
   954→   954→
   955→   955→  // authenticated reads
   956→   956→  if (!authenticated) return () => { throw unauthenticated(); };
   957→   957→  if (parts[1] === 'viewings' && parts[2] === 'mine') return () => resolveMyViewings();
   958→   958→  if (parts[1] === 'viewings' && parts[2] === 'assigned' && parts[3] === 'me') return () => resolveAssignedViewings();
   959→   959→  if (parts[1] === 'viewings' && parts[2] === 'dispatch-queue') return () => resolveDispatchQueue();
   960→   960→  if (parts[1] === 'viewings' && parts[2] === 'for-lister') return () => resolveListerViewings();
   961→   961→  // TASK 7-b (additive): filtered evidence lookup — must precede the
   962→   962→  // unfiltered introductions route below.
   963→   963→  if (parts[1] === 'viewings' && parts[2] === 'introductions' && hasIntroductionFilters(u.searchParams)) {
   964→   964→    return () => resolveIntroductionsFiltered(u.searchParams);
   965→   965→  }
   966→   966→  if (parts[1] === 'viewings' && parts[2] === 'introductions') return () => resolveIntroductions();
   967→   967→  if (parts[1] === 'viewings' && parts.length === 3) return () => resolveViewingDetail(parts[2]);
   968→   968→  if (parts[1] === 'identity' && parts[2] === 'me') return () => resolveIdentityMe();
   969→   969→  if (parts[1] === 'activity' && parts[2] === 'mine') return () => resolveActivityMine();
   970→   970→  if (parts[1] === 'deals' && parts.length === 2) return () => resolvePartyDeals();
   971→   971→  if (parts[1] === 'deals' && parts.length === 3) return () => resolveDealDetail(parts[2]);
   972→   972→  if (parts[1] === 'landlord' && parts[2] === 'financials') return () => resolveLandlordFinancials();
   973→   973→  if (parts[1] === 'admin' && parts[2] === 'launch-gate') return () => resolveLaunchGate();
   974→   974→  if (parts[1] === 'admin' && parts[2] === 'verification-queue') return () => resolveVerificationQueue();
   975→   975→  if (parts[1] === 'admin' && parts[2] === 'users') return () => resolveAdminUsers(u.searchParams);
   976→   976→  // TASK 7-b (additive): the deal queue's "Show only" filter passes ?status=
   977→   977→  // — must precede the unfiltered admin deals route below. The distribution
   978→   978→  // stays whole (it is the shape of the whole book); only the rows narrow.
   979→   979→  if (parts[1] === 'admin' && parts[2] === 'deals' && u.searchParams.has('status')) {
   980→   980→    return async () => {
   981→   981→      const status = u.searchParams.get('status') ?? '';
   982→   982→      const states = await resolveDealStates();
   983→   983→      const rows = states.rows.filter((r) => r.status === status);
   984→   984→      return { ...states, total: rows.length, rows };
   985→   985→    };
   986→   986→  }
   987→   987→  if (parts[1] === 'admin' && parts[2] === 'deals') return () => resolveDealStates();
   988→   988→  if (parts[1] === 'admin' && parts[2] === 'reconciliation') return () => resolveReconciliation();
   989→   989→  if (parts[1] === 'admin' && parts[2] === 'audit' && parts.length === 4) {
   990→   990→    return async () => toAuditEvents(await recentAuditEvents(50)).filter((e) => e.subjectRef === parts[3]);
   991→   991→  }
   992→   992→  if (parts[1] === 'admin' && parts[2] === 'audit') return async () => toAuditEvents(await recentAuditEvents(100));
   993→   993→  if (parts[1] === 'admin' && parts[2] === 'commission-rates') return () => resolveCommissionRates();
   994→   994→  if (parts[1] === 'admin' && parts[2] === 'config' && parts[4] === 'versions') return () => resolveConfigVersions(parts[3]);
   995→   995→
   996→   996→  throw new ApiError(404, 'UNKNOWN_PATH', `no in-process resolver for GET ${path}`);
   997→   997→}

  1017→  1017→  const method = init?.method ?? 'GET';

  1020→  1020→      500,
  1021→  1021→      'MUTATION_NOT_ALLOWED',
  1022→  1022→      'server-side mutations must go through /api/v1 route handlers via postJson()',
  1023→  1023→    );
  1024→  1024→  }
  1025→  1025→  return resolverFor(path, true)() as Promise<T>;
  1026→  1026→}
