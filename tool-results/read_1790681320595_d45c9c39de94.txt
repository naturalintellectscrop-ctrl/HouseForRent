   343→async function resolveListingsSearch(sp: URLSearchParams): Promise<SearchResponse> {
   344→  const filters: Parameters<typeof publicSearch>[0] = {};
   345→  const nid = sp.get('neighbourhoodId') ?? sp.get('neighbourhoodIds');
   346→  if (nid) filters.neighbourhoodIds = nid.split(',').filter(Boolean);
   347→  const minRent = sp.get('minRent');
   348→  if (minRent && /^[0-9]+$/.test(minRent)) filters.minRent = BigInt(minRent);
   349→  const maxRent = sp.get('maxRent');
   350→  if (maxRent && /^[0-9]+$/.test(maxRent)) filters.maxRent = BigInt(maxRent);
   351→  const bedrooms = sp.get('bedrooms');
   352→  if (bedrooms && /^[0-9]+$/.test(bedrooms)) filters.bedrooms = parseInt(bedrooms, 10);
   353→  const furnished = sp.get('furnished');
   354→  if (furnished) filters.furnished = furnished;
   355→  const propertyType = sp.get('propertyType');
   356→  if (propertyType) filters.propertyType = propertyType as never;
   357→  const q = sp.get('q');
   358→  if (q) filters.q = q;
   359→  const sort = sp.get('sort');
   360→  if (sort && ['fresh', 'rent_asc', 'rent_desc', 'newest'].includes(sort)) {
   361→    filters.sort = sort as never;
   362→  }
   363→  const limit = sp.get('limit');
   364→  if (limit && /^[0-9]+$/.test(limit)) filters.limit = parseInt(limit, 10);
   365→  const offset = sp.get('offset');
   366→  if (offset && /^[0-9]+$/.test(offset)) filters.offset = parseInt(offset, 10);
   367→  return publicSearch(filters);
   368→}

   370→async function resolveListingDetail(id: string) {
   371→  const detail = await publicDetail(id);
   372→  if (!detail) throw new ApiError(404, 'LISTING_NOT_FOUND', 'that listing does not exist or is not public');
   373→  // The officer's structured verdict on this home, when one exists — the
   374→  // same record the verification queue writes.
   375→  const lastConducted = await db.viewing.findFirst({
   376→    where: { listingId: id, status: 'conducted', fieldReport: { isNot: null } },
   377→    orderBy: { updatedAt: 'desc' },
   378→    include: { fieldReport: true },
   379→  });
   380→  const fieldConfirmed = lastConducted?.fieldReport
   381→    ? {
   382→        conditionRating: lastConducted.fieldReport.conditionRating as 'excellent' | 'good' | 'fair' | 'poor',
   383→        matchesListing: lastConducted.fieldReport.matchesListing,
   384→        isAvailable: lastConducted.fieldReport.isAvailable,
   385→        reportedAt: lastConducted.fieldReport.reportedAt.toISOString(),
   386→      }
   387→    : null;
   388→  // The caller's bookmark state, only when the caller is a signed-in tenant
   389→  // (saving is a tenant surface — anonymous visitors get no toggle).
   390→  const caller = await resolveSession();
   391→  const savedByCaller = caller?.role === 'tenant' ? await isSaved(caller.partyId, id) : null;
   392→  return {
   393→    ...detail,
   394→    neighbourhoodId: '',
   395→    geoLat: null,
   396→    geoLng: null,
   397→    amenities: detail.amenities.map((name, i) => ({ id: String(i), name })),
   398→    fieldConfirmed,
   399→    savedByCaller,
   400→  };
   401→}

   978→type Resolver = () => Promise<unknown>;
   979→
   980→/**
   981→ * Resolves an API path in-process. Route ORDER MATTERS: `/v1/listings/mine`
   982→ * must be tried before `/v1/listings/:id`, or "mine" would be read as an id.
   983→ */
   984→function resolverFor(path: string, authenticated: boolean): Resolver {
   985→  const u = new URL(path, 'http://in-process');
   986→  const parts = u.pathname.split('/').filter(Boolean); // ['v1', ...]
   987→
   988→  // public reads
   989→  if (parts[1] === 'listings' && parts.length === 2) return () => resolveListingsSearch(u.searchParams);
   990→  if (parts[1] === 'listings' && parts[2] === 'mine') return () => resolveMyListings();
   991→  // saved must precede /:id — same ordering rule as /mine above.
   992→  if (parts[1] === 'listings' && parts[2] === 'saved') return () => resolveSavedListings();
   993→  if (parts[1] === 'listings' && parts.length === 3) return () => resolveListingDetail(parts[2]);
   994→  if (parts[1] === 'listings' && parts[3] === 'agreement') return () => resolvePresentedTerms(parts[2]);
   995→  if (parts[1] === 'listings' && parts[3] === 'photos' && parts.length === 4) return () => resolveListingPhotos(parts[2]);
   996→  if (parts[1] === 'neighbourhoods') return () => resolveNeighbourhoods();
   997→  if (parts[1] === 'commission-rate') return () => resolveCommissionRate();
   998→
   999→  // authenticated reads
  1000→  if (!authenticated) return () => { throw unauthenticated(); };
  1001→  if (parts[1] === 'viewings' && parts[2] === 'mine') return () => resolveMyViewings();
  1002→  if (parts[1] === 'viewings' && parts[2] === 'assigned' && parts[3] === 'me') return () => resolveAssignedViewings();
  1003→  if (parts[1] === 'viewings' && parts[2] === 'dispatch-queue') return () => resolveDispatchQueue();
  1004→  if (parts[1] === 'viewings' && parts[2] === 'for-lister') return () => resolveListerViewings();
  1005→  // TASK 7-b (additive): filtered evidence lookup — must precede the
  1006→  // unfiltered introductions route below.
  1007→  if (parts[1] === 'viewings' && parts[2] === 'introductions' && hasIntroductionFilters(u.searchParams)) {
  1008→    return () => resolveIntroductionsFiltered(u.searchParams);
  1009→  }
  1010→  if (parts[1] === 'viewings' && parts[2] === 'introductions') return () => resolveIntroductions();
  1011→  if (parts[1] === 'viewings' && parts.length === 3) return () => resolveViewingDetail(parts[2]);
  1012→  if (parts[1] === 'identity' && parts[2] === 'me') return () => resolveIdentityMe();
  1013→  if (parts[1] === 'activity' && parts[2] === 'mine') return () => resolveActivityMine();
  1014→  if (parts[1] === 'deals' && parts.length === 2) return () => resolvePartyDeals();
  1015→  if (parts[1] === 'deals' && parts.length === 3) return () => resolveDealDetail(parts[2]);
  1016→  if (parts[1] === 'landlord' && parts[2] === 'financials') return () => resolveLandlordFinancials();
  1017→  // Task 10-b (additive): the mandate flow's reads. The decision POST is a
  1018→  // mutation and belongs to its route handler (which calls the service
  1019→  // directly), never to this GET-only dispatcher.
  1020→  if (parts[1] === 'landlord' && parts[2] === 'mandates') return () => resolveLandlordMandates();
  1021→  if (parts[1] === 'admin' && parts[2] === 'mandates') return () => resolveAdminMandates(u.searchParams);
  1022→  if (parts[1] === 'admin' && parts[2] === 'launch-gate') return () => resolveLaunchGate();
  1023→  if (parts[1] === 'admin' && parts[2] === 'verification-queue') return () => resolveVerificationQueue();
  1024→  if (parts[1] === 'admin' && parts[2] === 'users') return () => resolveAdminUsers(u.searchParams);
  1025→  // TASK 7-b (additive): the deal queue's "Show only" filter passes ?status=
  1026→  // — must precede the unfiltered admin deals route below. The distribution
  1027→  // stays whole (it is the shape of the whole book); only the rows narrow.
  1028→  if (parts[1] === 'admin' && parts[2] === 'deals' && u.searchParams.has('status')) {
  1029→    return async () => {
  1030→      const status = u.searchParams.get('status') ?? '';
  1031→      const states = await resolveDealStates();
  1032→      const rows = states.rows.filter((r) => r.status === status);
  1033→      return { ...states, total: rows.length, rows };
  1034→    };
  1035→  }
  1036→  if (parts[1] === 'admin' && parts[2] === 'deals') return () => resolveDealStates();
  1037→  if (parts[1] === 'admin' && parts[2] === 'reconciliation') return () => resolveReconciliation();
  1038→  if (parts[1] === 'admin' && parts[2] === 'audit' && parts.length === 4) {
  1039→    return async () => toAuditEvents(await recentAuditEvents(50)).filter((e) => e.subjectRef === parts[3]);
  1040→  }
  1041→  if (parts[1] === 'admin' && parts[2] === 'audit') return async () => toAuditEvents(await recentAuditEvents(100));
  1042→  if (parts[1] === 'admin' && parts[2] === 'commission-rates') return () => resolveCommissionRates();
  1043→  if (parts[1] === 'admin' && parts[2] === 'config' && parts[4] === 'versions') return () => resolveConfigVersions(parts[3]);
  1044→
  1045→  throw new ApiError(404, 'UNKNOWN_PATH', `no in-process resolver for GET ${path}`);
  1046→}

  1048→/**
  1049→ * An unauthenticated GET against the public API. The marketplace is
  1050→ * browsable without an account (Decision 3).
  1051→ */
  1052→export async function apiGet<T>(path: string, _opts: { revalidate?: number } = {}): Promise<T> {
  1053→  return resolverFor(path, false)() as Promise<T>;
  1054→}
  1055→
  1056→/**
  1057→ * Authenticated GET, in-process. Runs the role check here that the API's
  1058→ * guards would run — same rules, one implementation.
  1059→ *
  1060→ * MUTATIONS DO NOT BELONG HERE: any POST/PUT goes through `postJson()`
  1061→ * from '@/lib/client' in a CLIENT component, over real HTTP to /api/v1 —
  1062→ * this module is server-only (it imports the database), and every state
  1063→ * change must cross an API boundary.
  1064→ */
  1065→export async function api<T>(path: string, init?: { method?: string; body?: Record<string, unknown> }): Promise<T> {
  1066→  const method = init?.method ?? 'GET';
  1067→  if (method !== 'GET') {
  1068→    throw new ApiError(
  1069→      500,
  1070→      'MUTATION_NOT_ALLOWED',
  1071→      'server-side mutations must go through /api/v1 route handlers via postJson()',
  1072→    );
  1073→  }
  1074→  return resolverFor(path, true)() as Promise<T>;
