   340→async function resolveListingsSearch(sp: URLSearchParams): Promise<SearchResponse> {
   341→  const filters: Parameters<typeof publicSearch>[0] = {};
   342→  const nid = sp.get('neighbourhoodId') ?? sp.get('neighbourhoodIds');
   343→  if (nid) filters.neighbourhoodIds = nid.split(',').filter(Boolean);
   344→  const minRent = sp.get('minRent');
   345→  if (minRent && /^[0-9]+$/.test(minRent)) filters.minRent = BigInt(minRent);
   346→  const maxRent = sp.get('maxRent');
   347→  if (maxRent && /^[0-9]+$/.test(maxRent)) filters.maxRent = BigInt(maxRent);
   348→  const bedrooms = sp.get('bedrooms');
   349→  if (bedrooms && /^[0-9]+$/.test(bedrooms)) filters.bedrooms = parseInt(bedrooms, 10);
   350→  const furnished = sp.get('furnished');
   351→  if (furnished) filters.furnished = furnished;
   352→  const propertyType = sp.get('propertyType');
   353→  if (propertyType) filters.propertyType = propertyType as never;
   354→  const q = sp.get('q');
   355→  if (q) filters.q = q;
   356→  const sort = sp.get('sort');
   357→  if (sort && ['fresh', 'rent_asc', 'rent_desc', 'newest'].includes(sort)) {
   358→    filters.sort = sort as never;
   359→  }
   360→  const limit = sp.get('limit');
   361→  if (limit && /^[0-9]+$/.test(limit)) filters.limit = parseInt(limit, 10);
   362→  const offset = sp.get('offset');
   363→  if (offset && /^[0-9]+$/.test(offset)) filters.offset = parseInt(offset, 10);
   364→  return publicSearch(filters);
   365→}

   367→async function resolveListingDetail(id: string) {
   368→  const detail = await publicDetail(id);
   369→  if (!detail) throw new ApiError(404, 'LISTING_NOT_FOUND', 'that listing does not exist or is not public');
   370→  // The officer's structured verdict on this home, when one exists — the
   371→  // same record the verification queue writes.
   372→  const lastConducted = await db.viewing.findFirst({
   373→    where: { listingId: id, status: 'conducted', fieldReport: { isNot: null } },
   374→    orderBy: { updatedAt: 'desc' },
   375→    include: { fieldReport: true },
   376→  });
   377→  const fieldConfirmed = lastConducted?.fieldReport
   378→    ? {
   379→        conditionRating: lastConducted.fieldReport.conditionRating as 'excellent' | 'good' | 'fair' | 'poor',
   380→        matchesListing: lastConducted.fieldReport.matchesListing,
   381→        isAvailable: lastConducted.fieldReport.isAvailable,
   382→        reportedAt: lastConducted.fieldReport.reportedAt.toISOString(),
   383→      }
   384→    : null;
   385→  // The caller's bookmark state, only when the caller is a signed-in tenant
   386→  // (saving is a tenant surface — anonymous visitors get no toggle).
   387→  const caller = await resolveSession();
   388→  const savedByCaller = caller?.role === 'tenant' ? await isSaved(caller.partyId, id) : null;
   389→  return {
   390→    ...detail,
   391→    neighbourhoodId: '',
   392→    geoLat: null,
   393→    geoLng: null,
   394→    amenities: detail.amenities.map((name, i) => ({ id: String(i), name })),
   395→    fieldConfirmed,
   396→    savedByCaller,
   397→  };
   398→}

   570→async function resolveDispatchQueue(): Promise<DispatchQueue> {
   571→  const caller = await requireCaller();
   572→  if (caller.role !== 'admin' && caller.role !== 'foo') throw new ApiError(403, 'FORBIDDEN', 'staff surface');
   573→
   574→  const rows = await dispatchQueue();
   575→  const windowDays = await freshnessWindowDays();
   576→  const staleCutoff = new Date(Date.now() - windowDays * 24 * 60 * 60 * 1000);
   577→
   578→  const dispatchRows: DispatchRow[] = [];
   579→  for (const row of rows) {
   580→    const listing = await db.listing.findUnique({
   581→      where: { id: row.listingId },
   582→      include: { property: { include: { neighbourhood: true } } },
   583→    });
   584→    if (!listing) continue;
   585→    const blockedBy: string[] = [];
   586→    if (!listing.property.neighbourhood.inServiceArea) blockedBy.push('outside_service_area');
   587→    if (listing.verificationState !== 'verified') blockedBy.push('field_verification');
   588→    dispatchRows.push({
   589→      viewing: {
   590→        id: row.id,
   591→        listingId: row.listingId,

   934→type Resolver = () => Promise<unknown>;
   935→
   936→/**
   937→ * Resolves an API path in-process. Route ORDER MATTERS: `/v1/listings/mine`
   938→ * must be tried before `/v1/listings/:id`, or "mine" would be read as an id.
   939→ */
   940→function resolverFor(path: string, authenticated: boolean): Resolver {
   941→  const u = new URL(path, 'http://in-process');
   942→  const parts = u.pathname.split('/').filter(Boolean); // ['v1', ...]
   943→
   944→  // public reads
   945→  if (parts[1] === 'listings' && parts.length === 2) return () => resolveListingsSearch(u.searchParams);
   946→  if (parts[1] === 'listings' && parts[2] === 'mine') return () => resolveMyListings();
   947→  // saved must precede /:id — same ordering rule as /mine above.
   948→  if (parts[1] === 'listings' && parts[2] === 'saved') return () => resolveSavedListings();
   949→  if (parts[1] === 'listings' && parts.length === 3) return () => resolveListingDetail(parts[2]);
   950→  if (parts[1] === 'listings' && parts[3] === 'agreement') return () => resolvePresentedTerms(parts[2]);
   951→  if (parts[1] === 'listings' && parts[3] === 'photos' && parts.length === 4) return () => resolveListingPhotos(parts[2]);
   952→  if (parts[1] === 'neighbourhoods') return () => resolveNeighbourhoods();
   953→  if (parts[1] === 'commission-rate') return () => resolveCommissionRate();
   954→
   955→  // authenticated reads
   956→  if (!authenticated) return () => { throw unauthenticated(); };
   957→  if (parts[1] === 'viewings' && parts[2] === 'mine') return () => resolveMyViewings();
   958→  if (parts[1] === 'viewings' && parts[2] === 'assigned' && parts[3] === 'me') return () => resolveAssignedViewings();
   959→  if (parts[1] === 'viewings' && parts[2] === 'dispatch-queue') return () => resolveDispatchQueue();
   960→  if (parts[1] === 'viewings' && parts[2] === 'for-lister') return () => resolveListerViewings();
   961→  // TASK 7-b (additive): filtered evidence lookup — must precede the
   962→  // unfiltered introductions route below.
   963→  if (parts[1] === 'viewings' && parts[2] === 'introductions' && hasIntroductionFilters(u.searchParams)) {
   964→    return () => resolveIntroductionsFiltered(u.searchParams);
   965→  }
   966→  if (parts[1] === 'viewings' && parts[2] === 'introductions') return () => resolveIntroductions();
   967→  if (parts[1] === 'viewings' && parts.length === 3) return () => resolveViewingDetail(parts[2]);
   968→  if (parts[1] === 'identity' && parts[2] === 'me') return () => resolveIdentityMe();
   969→  if (parts[1] === 'activity' && parts[2] === 'mine') return () => resolveActivityMine();
   970→  if (parts[1] === 'deals' && parts.length === 2) return () => resolvePartyDeals();
   971→  if (parts[1] === 'deals' && parts.length === 3) return () => resolveDealDetail(parts[2]);
   972→  if (parts[1] === 'landlord' && parts[2] === 'financials') return () => resolveLandlordFinancials();
   973→  if (parts[1] === 'admin' && parts[2] === 'launch-gate') return () => resolveLaunchGate();
   974→  if (parts[1] === 'admin' && parts[2] === 'verification-queue') return () => resolveVerificationQueue();
   975→  if (parts[1] === 'admin' && parts[2] === 'users') return () => resolveAdminUsers(u.searchParams);
   976→  // TASK 7-b (additive): the deal queue's "Show only" filter passes ?status=
   977→  // — must precede the unfiltered admin deals route below. The distribution
   978→  // stays whole (it is the shape of the whole book); only the rows narrow.
   979→  if (parts[1] === 'admin' && parts[2] === 'deals' && u.searchParams.has('status')) {
   980→    return async () => {
   981→      const status = u.searchParams.get('status') ?? '';
   982→      const states = await resolveDealStates();
   983→      const rows = states.rows.filter((r) => r.status === status);
   984→      return { ...states, total: rows.length, rows };
   985→    };
   986→  }
   987→  if (parts[1] === 'admin' && parts[2] === 'deals') return () => resolveDealStates();
   988→  if (parts[1] === 'admin' && parts[2] === 'reconciliation') return () => resolveReconciliation();
   989→  if (parts[1] === 'admin' && parts[2] === 'audit' && parts.length === 4) {
   990→    return async () => toAuditEvents(await recentAuditEvents(50)).filter((e) => e.subjectRef === parts[3]);
   991→  }
   992→  if (parts[1] === 'admin' && parts[2] === 'audit') return async () => toAuditEvents(await recentAuditEvents(100));
   993→  if (parts[1] === 'admin' && parts[2] === 'commission-rates') return () => resolveCommissionRates();
   994→  if (parts[1] === 'admin' && parts[2] === 'config' && parts[4] === 'versions') return () => resolveConfigVersions(parts[3]);
   995→
   996→  throw new ApiError(404, 'UNKNOWN_PATH', `no in-process resolver for GET ${path}`);
   997→}

   999→/**
  1000→ * An unauthenticated GET against the public API. The marketplace is
  1001→ * browsable without an account (Decision 3).
  1002→ */
  1003→export async function apiGet<T>(path: string, _opts: { revalidate?: number } = {}): Promise<T> {
  1004→  return resolverFor(path, false)() as Promise<T>;
  1005→}

  1007→/**
  1008→ * Authenticated GET, in-process. Runs the role check here that the API's
  1009→ * guards would run — same rules, one implementation.
  1010→ *
  1011→ * MUTATIONS DO NOT BELONG HERE: any POST/PUT goes through `postJson()`
  1012→ * from '@/lib/client' in a CLIENT component, over real HTTP to /api/v1 —
  1013→ * this module is server-only (it imports the database), and every state
  1014→ * change must cross an API boundary.
  1015→ */
  1016→export async function api<T>(path: string, init?: { method?: string; body?: Record<string, unknown> }): Promise<T> {
  1017→  const method = init?.method ?? 'GET';
  1018→  if (method !== 'GET') {
  1019→    throw new ApiError(
  1020→      500,
  1021→      'MUTATION_NOT_ALLOWED',
  1022→      'server-side mutations must go through /api/v1 route handlers via postJson()',
  1023→    );
  1024→  }
  1025→  return resolverFor(path, true)() as Promise<T>;
  1026→}
