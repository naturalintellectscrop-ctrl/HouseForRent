   322→async function resolveListingsSearch(sp: URLSearchParams): Promise<SearchResponse> {
   323→  const filters: Parameters<typeof publicSearch>[0] = {};
   324→  const nid = sp.get('neighbourhoodId') ?? sp.get('neighbourhoodIds');
   325→  if (nid) filters.neighbourhoodIds = nid.split(',').filter(Boolean);
   326→  const minRent = sp.get('minRent');
   327→  if (minRent && /^[0-9]+$/.test(minRent)) filters.minRent = BigInt(minRent);
   328→  const maxRent = sp.get('maxRent');
   329→  if (maxRent && /^[0-9]+$/.test(maxRent)) filters.maxRent = BigInt(maxRent);
   330→  const bedrooms = sp.get('bedrooms');
   331→  if (bedrooms && /^[0-9]+$/.test(bedrooms)) filters.bedrooms = parseInt(bedrooms, 10);
   332→  const furnished = sp.get('furnished');
   333→  if (furnished) filters.furnished = furnished;
   334→  const propertyType = sp.get('propertyType');
   335→  if (propertyType) filters.propertyType = propertyType as never;
   336→  const q = sp.get('q');
   337→  if (q) filters.q = q;
   338→  const sort = sp.get('sort');
   339→  if (sort && ['fresh', 'rent_asc', 'rent_desc', 'newest'].includes(sort)) {
   340→    filters.sort = sort as never;
   341→  }
   342→  const limit = sp.get('limit');
   343→  if (limit && /^[0-9]+$/.test(limit)) filters.limit = parseInt(limit, 10);
   344→  const offset = sp.get('offset');
   345→  if (offset && /^[0-9]+$/.test(offset)) filters.offset = parseInt(offset, 10);
   346→  return publicSearch(filters);
   347→}

   349→async function resolveListingDetail(id: string) {
   350→  const detail = await publicDetail(id);
   351→  if (!detail) throw new ApiError(404, 'LISTING_NOT_FOUND', 'that listing does not exist or is not public');
   352→  // The officer's structured verdict on this home, when one exists — the
   353→  // same record the verification queue writes.

   355→    where: { listingId: id, status: 'conducted', fieldReport: { isNot: null } },
   356→    orderBy: { updatedAt: 'desc' },
   357→    include: { fieldReport: true },

   360→    ? {
   361→        conditionRating: lastConducted.fieldReport.conditionRating as 'excellent' | 'good' | 'fair' | 'poor',

   512→async function resolveDispatchQueue(): Promise<DispatchQueue> {
   513→  const caller = await requireCaller();
   514→  if (caller.role !== 'admin' && caller.role !== 'foo') throw new ApiError(403, 'FORBIDDEN', 'staff surface');
   515→
   516→  const rows = await dispatchQueue();
   517→  const windowDays = await freshnessWindowDays();
   518→  const staleCutoff = new Date(Date.now() - windowDays * 24 * 60 * 60 * 1000);
   519→
   520→  const dispatchRows: DispatchRow[] = [];
   521→  for (const row of rows) {
   522→    const listing = await db.listing.findUnique({
   523→      where: { id: row.listingId },
   524→      include: { property: { include: { neighbourhood: true } } },
   525→    });
   526→    if (!listing) continue;
   527→    const blockedBy: string[] = [];
   528→    if (!listing.property.neighbourhood.inServiceArea) blockedBy.push('outside_service_area');
   529→    if (listing.verificationState !== 'verified') blockedBy.push('field_verification');
   530→    dispatchRows.push({
   531→      viewing: {
   532→        id: row.id,
   533→        listingId: row.listingId,
   534→        tenantPartyId: '',
   535→        conductedByPartyId: null,
   536→        conductedByRole: 'foo',
   537→        scheduledFor: row.scheduledFor.toISOString(),
   538→        status: 'requested',
   539→        createdAt: row.requestedAt.toISOString(),
   540→        tenantName: row.tenantName,
   541→        neighbourhood: row.neighbourhood,
   542→        landmarkText: row.landmarkText,
   543→        bedrooms: row.bedrooms,
   544→        monthlyRent: row.monthlyRent,
   545→      },
   546→      listingId: row.listingId,
   547→      neighbourhood: row.neighbourhood,
   548→      inServiceArea: listing.property.neighbourhood.inServiceArea,
   549→      blockedBy,
   550→    });
   551→  }
   552→  void staleCutoff;
   553→
   554→  const officers = (await db.userAccount.findMany({
   555→    where: { role: 'foo', status: 'active' },
   556→    include: { party: { select: { displayName: true } } },
   557→  })).map(async (a) => ({
   558→    partyId: a.partyId,
   559→    displayName: a.party.displayName,
   560→    assignedCount: await db.viewing.count({
   561→      where: { conductedByPartyId: a.partyId, status: 'scheduled' },
   562→    }),
   563→  }));
   564→
   565→  return {
   566→    total: dispatchRows.length,
   567→    rows: dispatchRows,
   568→    officers: await Promise.all(officers),
   569→  };
   570→}

   572→async function resolveViewingDetail(viewingId: string): Promise<ViewingDetail> {
   573→  const caller = await requireCaller();
   574→  if (caller.role !== 'foo' && caller.role !== 'admin') throw new ApiError(403, 'FORBIDDEN', 'staff surface');
   575→  const detail = await getViewingDetail(viewingId);
   576→  if (detail.conductedByName && caller.role === 'foo' && detail.conductedByName !== caller.displayName) {
   577→    // Assigned to another officer: an admin may look, an officer may not.
   578→    throw new ApiError(403, 'NOT_ASSIGNED_OFFICER', 'this visit is assigned to another officer');
   579→  }
   580→  const fieldReport: FieldReport | null = detail.fieldReport
   581→    ? {
   582→        id: '',
   583→        viewingId: detail.id,
   584→        conditionRating: detail.fieldReport.conditionRating as never,
   585→        matchesListing: detail.fieldReport.matchesListing,
   586→        isAvailable: detail.fieldReport.isAvailable,
   587→        issuesText: detail.fieldReport.issuesText,
   588→        timingNote: detail.fieldReport.timingNote,
   589→        mediaAssetIds: [],
   590→        reportedAt: detail.fieldReport.reportedAt.toISOString(),
   591→      }
   592→    : null;
   593→  const introduction = detail.introductionRecord
   594→    ? {
   595→        id: detail.introductionRecord.id,

   709→/** The launch gate: verified-fresh live inventory against the configured bar. */
   710→async function resolveLaunchGate(): Promise<LaunchGate> {
   711→  const windowDays = await freshnessWindowDays();
   712→  const cutoff = new Date(Date.now() - windowDays * 24 * 60 * 60 * 1000);
   713→  const gateParam = await db.configParameter.findUnique({ where: { key: 'launch_gate_listings' } });
   714→  const gate = gateParam ? parseInt(gateParam.value, 10) || 12 : 12;
   715→
   716→  const live = await db.listing.count({
   717→    where: {
   718→      publicationState: 'live',
   719→      verificationState: 'verified',
   720→      availabilityConfirmedAt: { gte: cutoff },
   721→      property: { neighbourhood: { inServiceArea: true } },
   722→    },
   723→  });
   724→  const staleExcluded = await db.listing.count({
   725→    where: {
   726→      publicationState: 'live',
   727→      verificationState: 'verified',
   728→      property: { neighbourhood: { inServiceArea: true } },
   729→    },
   730→  });
   731→
   732→  return {
   733→    gate,
   734→    qualifying: live,
   735→    staleExcluded,
   736→    freshnessWindowDays: windowDays,
   737→    gateMet: live >= gate,
   738→    shortfall: Math.max(0, gate - live),
   739→    asOf: new Date().toISOString(),
   740→  };
   741→}

   844→type Resolver = () => Promise<unknown>;
   845→
   846→/**
   847→ * Resolves an API path in-process. Route ORDER MATTERS: `/v1/listings/mine`
   848→ * must be tried before `/v1/listings/:id`, or "mine" would be read as an id.
   849→ */
   850→function resolverFor(path: string, authenticated: boolean): Resolver {
   851→  const u = new URL(path, 'http://in-process');
   852→  const parts = u.pathname.split('/').filter(Boolean); // ['v1', ...]
   853→
   854→  // public reads
   855→  if (parts[1] === 'listings' && parts.length === 2) return () => resolveListingsSearch(u.searchParams);
   856→  if (parts[1] === 'listings' && parts[2] === 'mine') return () => resolveMyListings();
   857→  if (parts[1] === 'listings' && parts.length === 3) return () => resolveListingDetail(parts[2]);
   858→  if (parts[1] === 'listings' && parts[3] === 'agreement') return () => resolvePresentedTerms(parts[2]);
   859→  if (parts[1] === 'listings' && parts[3] === 'photos' && parts.length === 4) return () => resolveListingPhotos(parts[2]);
   860→  if (parts[1] === 'neighbourhoods') return () => resolveNeighbourhoods();
   861→  if (parts[1] === 'commission-rate') return () => resolveCommissionRate();
   862→
   863→  // authenticated reads
   864→  if (!authenticated) return () => { throw unauthenticated(); };
   865→  if (parts[1] === 'viewings' && parts[2] === 'mine') return () => resolveMyViewings();
   866→  if (parts[1] === 'viewings' && parts[2] === 'assigned' && parts[3] === 'me') return () => resolveAssignedViewings();
   867→  if (parts[1] === 'viewings' && parts[2] === 'dispatch-queue') return () => resolveDispatchQueue();
   868→  if (parts[1] === 'viewings' && parts[2] === 'introductions') return () => resolveIntroductions();
   869→  if (parts[1] === 'viewings' && parts.length === 3) return () => resolveViewingDetail(parts[2]);
   870→  if (parts[1] === 'identity' && parts[2] === 'me') return () => resolveIdentityMe();
   871→  if (parts[1] === 'deals' && parts.length === 2) return () => resolvePartyDeals();
   872→  if (parts[1] === 'deals' && parts.length === 3) return () => resolveDealDetail(parts[2]);
   873→  if (parts[1] === 'admin' && parts[2] === 'launch-gate') return () => resolveLaunchGate();
   874→  if (parts[1] === 'admin' && parts[2] === 'verification-queue') return () => resolveVerificationQueue();
   875→  if (parts[1] === 'admin' && parts[2] === 'deals') return () => resolveDealStates();
   876→  if (parts[1] === 'admin' && parts[2] === 'reconciliation') return () => resolveReconciliation();
   877→  if (parts[1] === 'admin' && parts[2] === 'audit' && parts.length === 4) {
   878→    return async () => toAuditEvents(await recentAuditEvents(50)).filter((e) => e.subjectRef === parts[3]);
   879→  }
   880→  if (parts[1] === 'admin' && parts[2] === 'audit') return async () => toAuditEvents(await recentAuditEvents(100));
   881→  if (parts[1] === 'admin' && parts[2] === 'commission-rates') return () => resolveCommissionRates();
   882→  if (parts[1] === 'admin' && parts[2] === 'config' && parts[4] === 'versions') return () => resolveConfigVersions(parts[3]);
   883→
   884→  throw new ApiError(404, 'UNKNOWN_PATH', `no in-process resolver for GET ${path}`);
   885→}

   887→/**
   888→ * An unauthenticated GET against the public API. The marketplace is
   889→ * browsable without an account (Decision 3).
   890→ */
   891→export async function apiGet<T>(path: string, _opts: { revalidate?: number } = {}): Promise<T> {
   892→  return resolverFor(path, false)() as Promise<T>;
   893→}
   894→
   895→/**
   896→ * Authenticated GET, in-process. Runs the role check here that the API's
   897→ * guards would run — same rules, one implementation.
   898→ *
   899→ * MUTATIONS DO NOT BELONG HERE: any POST/PUT goes through `postJson()`
   900→ * from '@/lib/client' in a CLIENT component, over real HTTP to /api/v1 —
   901→ * this module is server-only (it imports the database), and every state
   902→ * change must cross an API boundary.
   903→ */
   904→export async function api<T>(path: string, init?: { method?: string; body?: Record<string, unknown> }): Promise<T> {
   905→  const method = init?.method ?? 'GET';
   906→  if (method !== 'GET') {
   907→    throw new ApiError(
   908→      500,
   909→      'MUTATION_NOT_ALLOWED',
   910→      'server-side mutations must go through /api/v1 route handlers via postJson()',
   911→    );
   912→  }
   913→  return resolverFor(path, true)() as Promise<T>;
   914→}

   919→export async function requirePortalRole(allowed: readonly string[], returnTo?: string) {
   920→  const session = await resolveSession();
   921→  if (!session) redirect(returnTo ? `/login?next=${encodeURIComponent(returnTo)}` : '/login');

   923→  return session;
